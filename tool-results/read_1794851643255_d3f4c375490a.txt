   177→function parseNumber(raw: string): number | null {
   178→  const trimmed = raw.trim();
   179→  if (!trimmed) return null;
   180→  const parsed = Number(trimmed);
   181→  return Number.isFinite(parsed) ? parsed : null;
   182→}
   183→
   184→function buildCreatePayload(form: LeadFormValues): LeadCreatePayload {
   185→  const employees = parseNumber(form.employeeCount);
   186→  const dealValue = parseNumber(form.dealValue);
   187→  return {
   188→    companyName: form.companyName.trim(),
   189→    contactName: form.contactName.trim(),
   190→    industry: form.industry.trim() || undefined,
   191→    employeeCount: employees ?? undefined,
   192→    revenueBand: form.revenueBand || undefined,
   193→    country: form.country.trim() || undefined,
   194→    businessModel: form.businessModel || undefined,
   195→    email: form.email.trim() || undefined,
   196→    title: form.title.trim() || undefined,
   197→    phone: form.phone.trim() || undefined,
   198→    dealValue: dealValue ?? undefined,
   199→    source: form.source || undefined,
   200→  };
   201→}

   217→  };

   222→function ScoreCell({ value, percent = false }: { value: number | null; percent?: boolean }) {
   223→  if (value === null) {
   224→    return <span className="text-sm text-muted-foreground">{percent ? "n/a" : "—"}</span>;
   225→  }
   226→  const raw = percent ? Math.round(value * 100) : Math.round(value);
   227→  const clamped = Math.min(100, Math.max(0, raw));
   228→  return (
   229→    <div className="flex items-center gap-2">
   230→      <span className={cn("w-10 shrink-0 text-right text-sm font-medium tabular-nums", scoreColor(value))}>
   231→        {percent ? `${raw}%` : raw}
   232→      </span>
   233→      <Progress value={clamped} className="h-1 w-16 [&>div]:bg-emerald-500" />
   234→    </div>
   235→  );
   236→}

   519→function ImportDialog({
   520→  open,
   521→  onOpenChange,
   522→  onImported,
   523→}: {
   524→  open: boolean;
   525→  onOpenChange: (open: boolean) => void;
   526→  onImported: () => void;
   527→}) {
   528→  const [csv, setCsv] = useState("");
   529→  const [importing, setImporting] = useState(false);
   530→  const [result, setResult] = useState<ImportResult | null>(null);
   531→  const fileInputRef = useRef<HTMLInputElement>(null);
   532→
   533→  useEffect(() => {
   534→    if (open) setResult(null);
   535→  }, [open]);
   536→
   537→  function handleFile(event: ChangeEvent<HTMLInputElement>) {
   538→    const file = event.target.files?.[0];
   539→    if (!file) return;
   540→    const reader = new FileReader();
   541→    reader.onload = () => {
   542→      setCsv(typeof reader.result === "string" ? reader.result : "");
   543→      toast.success(`Loaded ${file.name}`);
   544→    };
   545→    reader.onerror = () => toast.error("Could not read the selected file.");
   546→    reader.readAsText(file);
   547→    event.target.value = "";
   548→  }
   549→
   550→  async function handleSubmit(event: FormEvent<HTMLFormElement>) {
   551→    event.preventDefault();
   552→    if (!csv.trim()) {
   553→      toast.error("Paste CSV content or choose a file first.");
   554→      return;
   555→    }
   556→    setImporting(true);
   557→    try {
   558→      const res = await api<ImportResult>("/api/leads/import", { method: "POST", body: { csv } });
   559→      setResult(res);
   560→      if (res.created > 0) {
   561→        toast.success(
   562→          `Imported ${res.created} lead${res.created === 1 ? "" : "s"}${res.skipped ? `, skipped ${res.skipped}` : ""}.`,
   563→        );
   564→      } else {
   565→        toast.warning(`No leads imported${res.skipped ? ` — ${res.skipped} skipped` : ""}.`);
   566→      }
   567→      if (res.errors && res.errors.length > 0) {
   568→        toast.warning(`${res.errors.length} row${res.errors.length === 1 ? "" : "s"} reported errors — see details.`);
   569→      }
   570→      onImported();
   571→      if (res.created > 0) setCsv("");
   572→    } catch (err) {
   573→      toast.error(err instanceof Error ? err.message : "Import failed. Please try again.");
   574→    } finally {
   575→      setImporting(false);
   576→    }
   577→  }
   578→
   579→  return (
   580→    <Dialog open={open} onOpenChange={onOpenChange}>
   581→      <DialogContent className="sm:max-w-xl">
   582→        <DialogHeader>
   583→          <DialogTitle>Import leads from CSV</DialogTitle>
   584→          <DialogDescription>
   585→            Paste CSV content or upload a file. One lead per row with a header row first.
   586→          </DialogDescription>
   587→        </DialogHeader>
   588→        <form onSubmit={handleSubmit} className="space-y-4" noValidate>
   589→          <div className="space-y-2">
   590→            <Label htmlFor="lm-import-csv">CSV content</Label>
   591→            <Textarea
   592→              id="lm-import-csv"
   593→              value={csv}
   594→              onChange={(e) => setCsv(e.target.value)}
   595→              rows={8}
   596→              className="font-mono text-xs"
   597→              placeholder={CSV_SAMPLE_HEADER}
   598→            />
   599→            <p className="rounded-md border bg-muted/40 px-3 py-2 text-xs text-muted-foreground">
   600→              Expected columns: <span className="font-mono">{CSV_SAMPLE_HEADER}</span>
   601→            </p>
   602→          </div>
   603→          <div className="flex flex-wrap items-center gap-3">
   604→            <input
   605→              ref={fileInputRef}
   606→              type="file"
   607→              id="lm-import-file"
   608→              accept=".csv,text/csv,text/plain"
   609→              className="sr-only"
   610→              aria-label="Choose CSV file"
   611→              onChange={handleFile}
   612→            />
   613→            <Button type="button" variant="outline" onClick={() => fileInputRef.current?.click()}>
   614→              <FileUp className="size-4" aria-hidden="true" />
   615→              Choose file
   616→            </Button>
   617→            {result && (
   618→              <p className="text-xs text-muted-foreground">
   619→                Created {result.created} · Skipped {result.skipped}
   620→              </p>
   621→            )}
   622→          </div>
   623→          {result && result.errors.length > 0 && (
   624→            <Alert variant="destructive">
   625→              <TriangleAlert className="size-4" />
   626→              <AlertTitle>{result.errors.length} row(s) were skipped</AlertTitle>
   627→              <AlertDescription>
   628→                <ul className="mt-1 max-h-32 list-disc space-y-0.5 overflow-y-auto pl-4 text-xs">
   629→                  {result.errors.map((err, i) => (
   630→                    <li key={i}>{err}</li>
   631→                  ))}
   632→                </ul>
   633→              </AlertDescription>
   634→            </Alert>
   635→          )}
   636→          <DialogFooter>
   637→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)} disabled={importing}>
   638→              Cancel
   639→            </Button>
   640→            <Button type="submit" disabled={importing || !csv.trim()}>
   641→              {importing ? "Importing..." : "Import leads"}
   642→            </Button>
   643→          </DialogFooter>
   644→        </form>
   645→      </DialogContent>
   646→    </Dialog>
   647→  );
   648→}

   658→  const [qualification, setQualification] = useState("ALL");

   660→  const [order, setOrder] = useState<SortOrder>("desc");
   661→  const [page, setPage] = useState(1);
   662→
   663→  const [items, setItems] = useState<LeadRowClient[]>([]);

   665→  const [industries, setIndustries] = useState<string[]>([]);

   669→  const [formOpen, setFormOpen] = useState(false);
   670→  const [editLead, setEditLead] = useState<LeadRowClient | null>(null);
   671→  const [importOpen, setImportOpen] = useState(false);
   672→
   673→  const reqIdRef = useRef(0);
   674→
   675→  // Debounce free-text search (350ms, no external deps).
   676→  useEffect(() => {
   677→    const timer = setTimeout(() => setSearchQuery(searchInput), SEARCH_DEBOUNCE_MS);
   678→    return () => clearTimeout(timer);
   679→  }, [searchInput]);
   680→
   681→  // Any filter or sort change resets pagination.

   683→    setPage(1);
   684→  }, [searchQuery, industry, qualification, sort, order]);
   685→
   686→  const load = useCallback(async () => {
   687→    const reqId = ++reqIdRef.current;
   688→    setLoading(true);
   689→    setError(null);
   690→    const params = new URLSearchParams();
   691→    if (searchQuery.trim()) params.set("query", searchQuery.trim());
   692→    if (industry !== "ALL") params.set("industry", industry);
   693→    if (qualification !== "ALL") params.set("qualification", qualification);
   694→    params.set("sort", sort);
   695→    params.set("order", order);
   696→    params.set("page", String(page));
   697→    params.set("pageSize", String(PAGE_SIZE));
   698→    try {
   699→      const res = await api<LeadsResponse>(`/api/leads?${params.toString()}`);
   700→      if (reqId !== reqIdRef.current) return;
   701→      setItems(res.items ?? []);
   702→      setTotal(res.total ?? 0);
   703→      if (res.facets && Array.isArray(res.facets.industries)) setIndustries(res.facets.industries);
   704→    } catch (err) {
   705→      if (reqId !== reqIdRef.current) return;
   706→      setError(err instanceof Error ? err.message : "Failed to load leads.");
   707→    } finally {
   708→      if (reqId === reqIdRef.current) setLoading(false);
   709→    }
   710→  }, [searchQuery, industry, qualification, sort, order, page]);

   713→    void load();
   714→  }, [load]);

   722→        setOrder("desc");

   744→                Export
   745→              </a>
   746→            </Button>
   747→            <Button
   748→              onClick={() => {
   749→                setEditLead(null);
   750→                setFormOpen(true);
   751→              }}
   752→            >
   753→              <Plus className="size-4" aria-hidden="true" />
   754→              New lead
   755→            </Button>
   756→          </>
   757→        }
   758→      />
