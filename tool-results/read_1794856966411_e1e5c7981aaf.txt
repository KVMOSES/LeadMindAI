    36→ * AnimatedAiChat — presentation layer for the LeadMind Copilot.
    37→ *
    38→ * This component renders the animated chat surface (message stream, typing
    39→ * indicator, tool activity, confirmation gate, composer with a `/` command
    40→ * palette). It contains **no AI logic of its own**: streaming, persistence,
    41→ * permissions and tool execution are owned by the existing backend and wired
    42→ * in through the callbacks below by `copilot-view.tsx`.

    44→ * Notes:
    45→ * - Sizes to its container (never `min-h-screen` / `w-screen`) so it lives
    46→ *   inside the application shell.
    47→ * - Framer Motion is used for entrance/exit transitions only; all animations
    48→ *   are disabled under `prefers-reduced-motion`.
    49→ * - No attachment controls: the current backend has no file-upload endpoint,
    50→ *   so none are rendered (no fake uploads).

   103→/**
   104→ * Slash commands are expanded into natural-language prompts before they reach
   105→ * the Copilot, so the assistant reliably triggers the matching capability
   106→ * regardless of how it interprets literal "/command" text.
   107→ */
   108→const COMMAND_EXPANSIONS: Record<string, string> = {
   109→  "/leads": "Search my leads",
   110→  "/qualify": "Qualify a prospect for me",
   111→  "/analyze": "Analyse a lead for me",
   112→  "/pipeline": "Summarise my current pipeline — counts, conversion estimates and distribution",
   113→  "/next-action": "What is the next best action I should take right now?",
   114→  "/insights": "Generate sales insights from my workspace data",
   115→  "/help": "What can you do? List your capabilities and the tools you use.",
   116→};
   117→
   118→export function expandSlashCommand(text: string): string {
   119→  const match = text.match(/^\/([a-zA-Z]+)\s*([\s\S]*)$/);
   120→  if (!match) return text;
   121→  const base = COMMAND_EXPANSIONS[`/${match[1].toLowerCase()}`];
   122→  if (!base) return text;
   123→  const rest = match[2].trim();
   124→  return rest ? `${base}. Details: ${rest}` : base;
   125→}

   266→                  className={cn(
   267→                    "flex h-7 w-7 shrink-0 items-center justify-center rounded-md",

   286→            </li>

   316→
   317→export function AnimatedAiChat({
   318→  messages,
   319→  loadingThread,
   320→  sending,
   321→  statusLine,
   322→  pending,
   323→  suggestions,
   324→  isDemo,
   325→  contextLead,
   326→  onSend,
   327→  onStop,
   328→  onRetry,
   329→  onResolvePending,
   330→  onOpenContextLead,
   331→  onClearContext,
   332→  headerActions,
   333→  className,
   334→}: AnimatedAiChatProps) {
   335→  const reduced = useReducedMotion();
   336→
   337→  const [input, setInput] = React.useState("");
   338→  const textareaRef = React.useRef<HTMLTextAreaElement>(null);
   339→  const scrollRef = React.useRef<HTMLDivElement>(null);
   340→  const bottomRef = React.useRef<HTMLDivElement>(null);
   341→  const [nearBottom, setNearBottom] = React.useState(true);
   342→
   343→  // ── Command palette state ────────────────────────────────────────────────
   344→  const paletteOpen = input.startsWith("/") && !sending;
   345→  const paletteQuery = paletteOpen ? input.slice(1).split(/\s/)[0]?.toLowerCase() ?? "" : "";
   346→  const filteredCommands = React.useMemo(() => {
   347→    if (!paletteOpen) return [];
   348→    if (!paletteQuery) return COPILOT_COMMANDS;
   349→    return COPILOT_COMMANDS.filter((c) => c.command.slice(1).startsWith(paletteQuery));
   350→  }, [paletteOpen, paletteQuery]);
   351→  const [activeCmdIndex, setActiveCmdIndex] = React.useState(0);
   352→
   353→  React.useEffect(() => {
   354→    setActiveCmdIndex(0);
   355→  }, [paletteQuery]);
   356→
   357→  // ── Auto-resizing composer ───────────────────────────────────────────────
   358→  React.useLayoutEffect(() => {
   359→    const el = textareaRef.current;
   360→    if (!el) return;
   361→    el.style.height = "auto";
   362→    el.style.height = `${Math.min(el.scrollHeight, 180)}px`;
   363→  }, [input]);
   364→
   365→  // ── Scroll behaviour: follow only when the user is already at the bottom ──
   366→  const handleScroll = React.useCallback(() => {
   367→    const el = scrollRef.current;
   368→    if (!el) return;
   369→    setNearBottom(el.scrollHeight - el.scrollTop - el.clientHeight < 80);
   370→  }, []);
   371→
   372→  React.useEffect(() => {
   373→    if (nearBottom) {
   374→      bottomRef.current?.scrollIntoView({ behavior: reduced ? "auto" : "smooth", block: "end" });
   375→    }
   376→  }, [messages, statusLine, nearBottom, reduced]);
   377→
   378→  const scrollToLatest = () => {
   379→    bottomRef.current?.scrollIntoView({ behavior: reduced ? "auto" : "smooth", block: "end" });
   380→  };
   381→
   382→  const selectCommand = (cmd: CopilotCommand) => {
   383→    if (cmd.send) {
   384→      setInput("");
   385→      onSend(cmd.send);
   386→      textareaRef.current?.focus();
   387→      return;
   388→    }
   389→    setInput(cmd.template);
   390→    requestAnimationFrame(() => {
   391→      textareaRef.current?.focus();
   392→      const el = textareaRef.current;
   393→      if (el) el.selectionStart = el.selectionEnd = el.value.length;
   394→    });
   395→  };
   396→
   397→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   398→    const trimmedInput = input.trim();
   399→    // Exact command match (e.g. "/pipeline") → Enter sends instead of re-selecting.
   400→    const exactCommand =
   401→      paletteOpen &&
   402→      trimmedInput.length > 1 &&
   403→      filteredCommands.some((c) => c.command === trimmedInput);
   404→
   405→    if (paletteOpen && filteredCommands.length > 0) {
   406→      if (e.key === "ArrowDown") {
   407→        e.preventDefault();
   408→        setActiveCmdIndex((i) => (i + 1) % filteredCommands.length);
   409→        return;
   410→      }
   411→      if (e.key === "ArrowUp") {
   412→        e.preventDefault();
   413→        setActiveCmdIndex((i) => (i - 1 + filteredCommands.length) % filteredCommands.length);
   414→        return;
   415→      }
   416→      if ((e.key === "Enter" || e.key === "Tab") && !exactCommand) {
   417→        e.preventDefault();
   418→        selectCommand(filteredCommands[activeCmdIndex]);
   419→        return;
   420→      }
   421→      if (e.key === "Escape") {
   422→        e.preventDefault();
   423→        setInput("");
   424→        return;
   425→      }
   426→    }
   427→    if (e.key === "Enter" && !e.shiftKey) {
   428→      e.preventDefault();
   429→      if (trimmedInput) {
   430→        setInput("");
   431→        onSend(expandSlashCommand(trimmedInput));
   432→      }
   433→    }
   434→  };
   435→
   436→  const lastMsg = messages[messages.length - 1];
   437→  const lastAssistantErrored = lastMsg?.role === "ASSISTANT" && !lastMsg.streaming && lastMsg.content.startsWith("**Error:**");
   438→  const canRetry = !sending && messages.some((m) => m.role === "USER");
   439→
   440→  return (
   441→    <div className={cn("flex h-full min-h-0 w-full min-w-0 flex-col", className)} data-chat-surface="">
   442→      {/* ── Chat toolbar ──────────────────────────────────────────────────── */}
   443→      <div className="relative z-10 flex items-center justify-between gap-2 border-b border-border/70 px-4 py-2.5">
   444→        <div className="flex min-w-0 items-center gap-2">
   445→          <LogoMark size={26} />
   446→          <div className="min-w-0">
   447→            <p className="truncate font-display text-sm font-bold">LeadMind Copilot</p>
   448→            {contextLead ? (
   449→              <p className="truncate text-xs text-muted-foreground">
   450→                Context:{" "}
   451→                <button type="button" className="font-medium text-[var(--lm-brand)] hover:underline" onClick={onOpenContextLead}>
   452→                  {contextLead.label}
   453→                </button>
   454→                <button type="button" className="ml-2 text-muted-foreground hover:underline" onClick={onClearContext}>
   455→                  clear
   456→                </button>
   457→              </p>
   458→            ) : (
   459→              <p className="truncate text-xs text-muted-foreground">
   460→                {isDemo ? "Local demo assistant — deterministic, live workspace data" : "Connected to external LLM"}
   461→              </p>
   462→            )}
   463→          </div>
   464→        </div>
   465→        <div className="flex shrink-0 items-center gap-1.5">
   466→          <span
   467→            className={cn(
   468→              "inline-flex items-center gap-1.5 rounded-full border px-2 py-0.5 text-[10px] font-medium",
   469→              isDemo
   470→                ? "border-amber-500/40 text-amber-600 dark:text-amber-400"
   471→                : "border-[var(--lm-brand)]/40 text-[var(--lm-brand)]",
   472→            )}
   473→          >
   474→            <span className={cn("h-1.5 w-1.5 rounded-full", isDemo ? "bg-amber-500" : "bg-[var(--lm-brand)] lm-pulse-dot")} aria-hidden="true" />
   475→            {isDemo ? "DEMO" : "LIVE LLM"}
   476→          </span>
   477→          {canRetry && (
   478→            <Button
   479→              variant="ghost"
   480→              size="icon"
   481→              className="h-8 w-8"
   482→              onClick={onRetry}
   483→              disabled={sending}
   484→              aria-label="Regenerate last answer"
   485→              title="Regenerate last answer"
   486→            >
   487→              <RefreshCcw className="h-4 w-4" />
   488→            </Button>
   489→          )}
   490→          {headerActions}
   491→        </div>
   492→      </div>
   493→
   494→      {/* ── Message stream ────────────────────────────────────────────────── */}
   495→      <div className="relative min-h-0 flex-1">
   496→        <div ref={scrollRef} onScroll={handleScroll} className="lm-scroll absolute inset-0 overflow-y-auto">
   497→          <div className="relative z-10 space-y-4 px-4 py-4">
   498→            {loadingThread ? (
   499→              <div className="space-y-3" aria-label="Loading conversation">
   500→                {[0, 1, 2].map((i) => (
   501→                  <motion.div
   502→                    key={i}
   503→                    className={cn("h-16 rounded-2xl border border-border/60 bg-muted/40", i % 2 === 1 && "ml-auto w-3/4")}
   504→                    style={{ width: i % 2 === 1 ? undefined : `${55 + i * 8}%` }}
   505→                    initial={reduced ? false : { opacity: 0 }}
   506→                    animate={{ opacity: [0.4, 0.85, 0.4] }}
   507→                    transition={{ duration: 1.6, repeat: Infinity, delay: i * 0.2 }}
   508→                  />
   509→                ))}
   510→              </div>
   511→            ) : messages.length === 0 ? (
   512→              <motion.div
   513→                className="flex flex-col items-center py-12 text-center"
   514→                initial={reduced ? false : { opacity: 0, y: 14 }}
   515→                animate={{ opacity: 1, y: 0 }}
   516→                transition={{ duration: 0.4 }}
   517→              >
   518→                <div className="relative mb-4">
   519→                  <div
   520→                    aria-hidden="true"
   521→                    className="absolute inset-0 -z-10 rounded-full blur-2xl"

   545→                        {s}
   546→                      </motion.button>

   548→                  </div>

   550→              </motion.div>

   552→              <AnimatePresence initial={false} mode="popLayout">

   555→                    key={m.id}
   556→                    layout={!reduced}
   557→                    variants={MESSAGE_VARIANTS}
   558→                    initial={reduced ? false : "hidden"}
   559→                    animate="show"
   560→                    exit="exit"
   561→                    className={cn("flex gap-3", m.role === "USER" ? "justify-end" : "justify-start")}
   562→                  >
   563→                    {m.role === "ASSISTANT" && (
   564→                      <div className="mt-1 flex h-7 w-7 shrink-0 items-center justify-center rounded-md bg-[var(--lm-brand-soft)] ring-1 ring-[var(--lm-brand)]/25">
   565→                        <LogoMark size={18} badge={false} />
   566→                      </div>

   568→                    <div className={cn("min-w-0 max-w-[85%] space-y-2 sm:max-w-[78%]", m.role === "USER" && "flex flex-col items-end")}>
   569→                      {m.toolTraces.length > 0 && <ToolTracePanel traces={m.toolTraces} />}
   570→                      <div
   571→                        className={cn(
   572→                          "rounded-2xl px-4 py-3 text-sm",
   573→                          m.role === "USER"
   574→                            ? "rounded-br-md bg-[linear-gradient(135deg,var(--lm-brand),color-mix(in_oklab,var(--lm-brand),var(--lm-info)_35%))] text-white shadow-[0_6px_18px_-8px_var(--lm-brand)]"

   584→                              <Markdown content={m.content} />

   588→                          </div>

   591→                      <div className="flex items-center gap-2 px-1 text-[10px] text-muted-foreground">

   605→                            <Copy className="h-3 w-3" aria-hidden="true" /> copy

   610→                            type="button"
   611→                            className="ml-auto inline-flex items-center gap-1 rounded px-1 font-medium text-[var(--lm-brand)] transition-colors hover:underline"

   613→                            disabled={sending}
   614→                          >
   615→                            <RefreshCcw className="h-3 w-3" aria-hidden="true" /> Try again

   618→                      </div>
   619→                    </div>
   620→                    {m.role === "USER" && (
   621→                      <div className="mt-1 flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-muted">
   622→                        <User className="h-3.5 w-3.5 text-muted-foreground" aria-hidden="true" />
   623→                      </div>
   624→                    )}
   625→                  </motion.div>
   626→                ))}
   627→              </AnimatePresence>
   628→            )}
   629→
   630→            {/* Confirmation gate (existing pending-action protocol) */}
   631→            <AnimatePresence>
   632→              {pending && (
   633→                <motion.div
   634→                  key="pending"
   635→                  initial={reduced ? false : { opacity: 0, y: 10, scale: 0.98 }}
   636→                  animate={{ opacity: 1, y: 0, scale: 1 }}
   637→                  exit={{ opacity: 0, scale: 0.98 }}
   638→                  transition={{ duration: 0.22 }}
   639→                  className="mx-auto w-full max-w-lg overflow-hidden rounded-xl border border-amber-500/40 bg-amber-50/80 backdrop-blur-md dark:bg-amber-950/30"

   646→                        <ShieldQuestion className="h-4 w-4 text-amber-600 dark:text-amber-400" aria-hidden="true" />
   647→                      </span>
   648→                      Confirmation required
   649→                    </div>
   650→                    <p className="text-sm leading-relaxed text-muted-foreground">{pending.summary}</p>
   651→                    <div className="flex flex-wrap gap-2">
   652→                      <Button
   653→                        size="sm"
   654→                        onClick={() => onResolvePending("confirm")}
   655→                        className="bg-amber-600 text-white hover:bg-amber-500"
   656→                      >
   657→                        <Check className="mr-1 h-3.5 w-3.5" aria-hidden="true" /> Confirm &amp; execute
   658→                      </Button>
   659→                      <Button size="sm" variant="ghost" onClick={() => onResolvePending("cancel")}>
   660→                        <X className="mr-1 h-3.5 w-3.5" aria-hidden="true" /> Cancel
   661→                      </Button>
   662→                    </div>
   663→                  </div>
   664→                </motion.div>
   665→              )}
   666→            </AnimatePresence>
   667→
   668→            {/* Assistant status line with animated indicator */}
   669→            <AnimatePresence>
   670→              {statusLine && (
   671→                <motion.div
   672→                  key="status"
   673→                  initial={reduced ? false : { opacity: 0, y: 6 }}
   674→                  animate={{ opacity: 1, y: 0 }}
   675→                  exit={{ opacity: 0 }}
   676→                  className="flex items-center gap-2.5 pl-10 text-xs text-muted-foreground"

   678→                >

   682→                    <TypingDots />
   683→                  )}
   684→                  <span>{statusLine}</span>
   685→                </motion.div>
   686→              )}
   687→            </AnimatePresence>
   688→
   689→            <div ref={bottomRef} aria-hidden="true" />
   690→          </div>
   691→        </div>
   692→
   693→        {/* Scroll-to-latest */}
   694→        <AnimatePresence>
   695→          {!nearBottom && messages.length > 0 && (
   696→            <motion.div
   697→              initial={reduced ? false : { opacity: 0, y: 8 }}
   698→              animate={{ opacity: 1, y: 0 }}
   699→              exit={{ opacity: 0, y: 8 }}
   700→              className="absolute bottom-3 left-1/2 z-20 -translate-x-1/2"

   703→                size="sm"

   707→                aria-label="Scroll to latest messages"
   708→              >
   709→                <ArrowDown className="mr-1 h-3.5 w-3.5" aria-hidden="true" />
   710→                Latest
   711→              </Button>
   712→            </motion.div>
   713→          )}
   714→        </AnimatePresence>
   715→      </div>
   716→
   717→      {/* ── Composer ──────────────────────────────────────────────────────── */}
   718→      <div className="relative z-10 border-t border-border/70 p-3">
   719→        <div className="relative">
   720→          <CommandPalette
   721→            visible={paletteOpen}
   722→            activeIndex={activeCmdIndex}
   723→            items={filteredCommands}
   724→            onSelect={selectCommand}
   725→          />
   726→
   727→          <div className="flex items-end gap-2 rounded-xl border border-border/70 bg-card/70 p-2 shadow-sm backdrop-blur-md transition-[border-color,box-shadow] focus-within:border-[var(--lm-brand)]/40 focus-within:shadow-[0_0_0_3px_var(--lm-brand-soft)]">
   728→            <Textarea
   729→              ref={textareaRef}
   730→              value={input}
   731→              onChange={(e) => setInput(e.target.value)}

   733→              placeholder="Ask about leads, scores, predictions, actions…  Type / for commands"
   734→              rows={1}

   736→              className="max-h-[180px] min-h-[2.25rem] flex-1 resize-none border-0 bg-transparent px-2 py-1.5 text-sm shadow-none focus-visible:ring-0"

   738→            />
   739→            {sending ? (

   741→                variant="destructive"
