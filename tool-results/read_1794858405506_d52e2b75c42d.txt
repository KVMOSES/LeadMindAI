    35→/**
    36→ * AnimatedAiChat — presentation layer for the LeadMind Copilot.
    37→ *
    38→ * This component renders the animated chat surface (message stream, typing
    39→ * indicator, tool activity, confirmation gate, composer with a `/` command
    40→ * palette). It contains **no AI logic of its own**: streaming, persistence,
    41→ * permissions and tool execution are owned by the existing backend and wired
    42→ * in through the callbacks below by `copilot-view.tsx`.
    43→ *
    44→ * Notes:
    45→ * - Sizes to its container (never `min-h-screen` / `w-screen`) so it lives
    46→ *   inside the application shell.
    47→ * - Framer Motion is used for entrance/exit transitions only; all animations
    48→ *   are disabled under `prefers-reduced-motion`.
    49→ * - No attachment controls: the current backend has no file-upload endpoint,
    50→ *   so none are rendered (no fake uploads).
    51→ * - Composer enforces the real backend limit (4000 characters).
    52→ */

   107→/**
   108→ * Slash commands are expanded into natural-language prompts before they reach
   109→ * the Copilot, so the assistant reliably triggers the matching capability
   110→ * regardless of how it interprets literal "/command" text.
   111→ */
   112→const COMMAND_EXPANSIONS: Record<string, string> = {
   113→  "/leads": "Search my leads",
   114→  "/qualify": "Qualify a prospect for me",
   115→  "/analyze": "Analyse a lead for me",
   116→  "/pipeline": "Summarise my current pipeline — counts, conversion estimates and distribution",
   117→  "/next-action": "What is the next best action I should take right now?",
   118→  "/insights": "Generate sales insights from my workspace data",
   119→  "/help": "What can you do? List your capabilities and the tools you use.",
   120→};
   121→
   122→export function expandSlashCommand(text: string): string {
   123→  const match = text.match(/^\/([a-zA-Z]+)\s*([\s\S]*)$/);
   124→  if (!match) return text;
   125→  const base = COMMAND_EXPANSIONS[`/${match[1].toLowerCase()}`];
   126→  if (!base) return text;
   127→  const rest = match[2].trim();
   128→  return rest ? `${base}. Details: ${rest}` : base;
   129→}

   338→  className?: string;
   339→}
   340→
   341→export function AnimatedAiChat({
   342→  messages,
   343→  loadingThread,
   344→  sending,
   345→  statusLine,
   346→  pending,
   347→  suggestions,
   348→  isDemo,
   349→  contextLead,
   350→  conversationTitle,
   351→  onSend,
   352→  onStop,
   353→  onRetry,
   354→  onResolvePending,
   355→  onOpenContextLead,
   356→  onClearContext,
   357→  headerActions,
   358→  className,
   359→}: AnimatedAiChatProps) {
   360→  const reduced = useReducedMotion();
   361→
   362→  const [input, setInput] = React.useState("");
   363→  const textareaRef = React.useRef<HTMLTextAreaElement>(null);
   364→  const scrollRef = React.useRef<HTMLDivElement>(null);
   365→  const bottomRef = React.useRef<HTMLDivElement>(null);
   366→  const [nearBottom, setNearBottom] = React.useState(true);
   367→
   368→  // ── Command palette state ────────────────────────────────────────────────
   369→  const paletteOpen = input.startsWith("/") && !sending;
   370→  const paletteQuery = paletteOpen ? input.slice(1).split(/\s/)[0]?.toLowerCase() ?? "" : "";
   371→  const filteredCommands = React.useMemo(() => {
   372→    if (!paletteOpen) return [];
   373→    if (!paletteQuery) return COPILOT_COMMANDS;
   374→    return COPILOT_COMMANDS.filter((c) => c.command.slice(1).startsWith(paletteQuery));
   375→  }, [paletteOpen, paletteQuery]);
   376→  const [activeCmdIndex, setActiveCmdIndex] = React.useState(0);
   377→
   378→  React.useEffect(() => {
   379→    setActiveCmdIndex(0);
   380→  }, [paletteQuery]);
   381→
   382→  // ── Auto-resizing composer ───────────────────────────────────────────────
   383→  React.useLayoutEffect(() => {
   384→    const el = textareaRef.current;
   385→    if (!el) return;
   386→    const resize = () => {
   387→      el.style.height = "auto";
   388→      el.style.height = `${Math.min(el.scrollHeight, 180)}px`;
   389→    };
   390→    resize();
   391→    // Re-measure when the viewport changes so the placeholder never clips.
   392→    window.addEventListener("resize", resize);
   393→    return () => window.removeEventListener("resize", resize);
   394→  }, [input]);
   395→
   396→  // ── Scroll behaviour: follow only when the user is already at the bottom ──
   397→  const handleScroll = React.useCallback(() => {
   398→    const el = scrollRef.current;
   399→    if (!el) return;
   400→    setNearBottom(el.scrollHeight - el.scrollTop - el.clientHeight < 80);
   401→  }, []);
   402→
   403→  React.useEffect(() => {
   404→    if (nearBottom) {
   405→      bottomRef.current?.scrollIntoView({ behavior: reduced ? "auto" : "smooth", block: "end" });
   406→    }
   407→  }, [messages, statusLine, nearBottom, reduced]);
   408→
   409→  const scrollToLatest = () => {
   410→    bottomRef.current?.scrollIntoView({ behavior: reduced ? "auto" : "smooth", block: "end" });
   411→  };
   412→
   413→  const submitInput = () => {
   414→    const trimmed = input.trim();
   415→    if (!trimmed || sending) return;
   416→    setInput("");
   417→    onSend(expandSlashCommand(trimmed));
   418→  };
   419→
   420→  const selectCommand = (cmd: CopilotCommand) => {
   421→    if (cmd.send) {
   422→      setInput("");
   423→      onSend(cmd.send);
   424→      textareaRef.current?.focus();
   425→      return;
   426→    }
   427→    setInput(cmd.template);
   428→    requestAnimationFrame(() => {
   429→      textareaRef.current?.focus();
   430→      const el = textareaRef.current;
   431→      if (el) el.selectionStart = el.selectionEnd = el.value.length;
   432→    });
   433→  };
   434→
   435→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   436→    const trimmedInput = input.trim();
   437→    // Exact command match (e.g. "/pipeline") → Enter sends instead of re-selecting.
   438→    const exactCommand =
   439→      paletteOpen &&
   440→      trimmedInput.length > 1 &&
   441→      filteredCommands.some((c) => c.command === trimmedInput);
   442→
   443→    if (paletteOpen && filteredCommands.length > 0) {
   444→      if (e.key === "ArrowDown") {
   445→        e.preventDefault();
   446→        setActiveCmdIndex((i) => (i + 1) % filteredCommands.length);
   447→        return;
   448→      }
   449→      if (e.key === "ArrowUp") {
   450→        e.preventDefault();
   451→        setActiveCmdIndex((i) => (i - 1 + filteredCommands.length) % filteredCommands.length);
   452→        return;
   453→      }
   454→      if ((e.key === "Enter" || e.key === "Tab") && !exactCommand) {
   455→        e.preventDefault();
   456→        selectCommand(filteredCommands[activeCmdIndex]);
   457→        return;
   458→      }
   459→      if (e.key === "Escape") {
   460→        e.preventDefault();
   461→        setInput("");
   462→        return;
   463→      }
   464→    }
   465→    if (e.key === "Enter" && !e.shiftKey) {
   466→      e.preventDefault();
   467→      submitInput();
   468→    }
   469→  };
   470→
   471→  const lastMsg = messages[messages.length - 1];
   472→  const lastAssistantErrored = lastMsg?.role === "ASSISTANT" && !lastMsg.streaming && lastMsg.content.startsWith("**Error:**");
   473→  const canRetry = !sending && messages.some((m) => m.role === "USER");
   474→  const nearLimit = input.length > MAX_MESSAGE_LENGTH - 400;
   475→
   476→  return (
   477→    <div className={cn("flex h-full min-h-0 w-full min-w-0 flex-col", className)} data-chat-surface="">
   478→      {/* ── Chat toolbar ──────────────────────────────────────────────────── */}
   479→      <div className="relative z-10 flex items-center justify-between gap-3 border-b border-border/70 px-4 py-2.5 sm:px-5">
   480→        <div className="flex min-w-0 items-center gap-2.5">
   481→          <LogoMark size={28} />
   482→          <div className="min-w-0">
   483→            <p className="truncate font-display text-sm font-bold leading-tight">LeadMind Copilot</p>
   484→            {conversationTitle ? (
   485→              <p className="truncate text-xs text-muted-foreground" title={conversationTitle}>
   486→                {conversationTitle}
   487→              </p>
   488→            ) : contextLead ? (
   489→              <p className="truncate text-xs text-muted-foreground">
   490→                Context:{" "}
   491→                <button type="button" className="font-medium text-[var(--lm-brand)] hover:underline" onClick={onOpenContextLead}>
   492→                  {contextLead.label}
   493→                </button>
   494→                <button type="button" className="ml-2 text-muted-foreground hover:underline" onClick={onClearContext}>
   495→                  clear
   496→                </button>
   497→              </p>
   498→            ) : (
   499→              <p className="truncate text-xs text-muted-foreground">
   500→                {isDemo ? "Local demo assistant — deterministic, live workspace data" : "Connected to external LLM"}
   501→              </p>
   502→            )}
   503→          </div>
   504→        </div>
   505→        <div className="flex shrink-0 items-center gap-1.5">
   506→          {contextLead && conversationTitle && (
   507→            <button
   508→              type="button"
   509→              onClick={onOpenContextLead}
   510→              className="hidden max-w-[16rem] items-center truncate rounded-full border border-[var(--lm-brand)]/30 bg-[var(--lm-brand-soft)] px-2.5 py-0.5 text-[10.5px] font-medium text-[var(--lm-brand)] hover:border-[var(--lm-brand)]/60 xl:inline-flex"
   511→              title={`Open lead: ${contextLead.label}`}
   512→            >
   513→              <User className="mr-1 size-3" aria-hidden="true" />
   514→              {contextLead.label}
   515→            </button>
   516→          )}
   517→          <span
   518→            className={cn(
   519→              "inline-flex items-center gap-1.5 rounded-full border px-2 py-0.5 text-[10px] font-medium",
   520→              isDemo
   521→                ? "border-amber-500/40 text-amber-600 dark:text-amber-400"
   522→                : "border-[var(--lm-brand)]/40 text-[var(--lm-brand)]",
   523→            )}
   524→          >
