    83→// ── Formatting helpers ───────────────────────────────────────────────────────
    84→
    85→function prettifyAction(action: string): string {
    86→  const s = action.replaceAll("_", " ").toLowerCase();
    87→  return s.charAt(0).toUpperCase() + s.slice(1);
    88→}
    89→
    90→function truncate(text: string, max: number): string {
    91→  return text.length > max ? `${text.slice(0, max).trimEnd()}…` : text;
    92→}
    93→
    94→function fmtScore(value: number | null): string {
    95→  return value === null ? "—" : value.toFixed(1);
    96→}
    97→
    98→function fmtMetric(value: number | null): string {
    99→  return value === null ? "—" : value.toFixed(2);
   100→}
   101→
   102→function fmtConversion(rate: number | null): string {
   103→  return rate === null ? "—" : `${Math.round(rate * 100)}%`;
   104→}
   105→
   106→function activityDotColor(type: string): string {
   107→  if (type === "EVENT") return "bg-[var(--lm-info)]";
   108→  if (type === "OUTCOME") return "bg-[var(--lm-brand)]";
   109→  return "bg-zinc-400";
   110→}

   114→function PanelCard({
   115→  title,
   116→  subtitle,
   117→  actions,
   118→  className,
   119→  children,
   120→  spotlight = false,
   121→  spotlightColor,
   122→}: {
   123→  title: string;
   124→  subtitle?: string;
   125→  actions?: ReactNode;
   126→  className?: string;
   127→  children: ReactNode;
   128→  spotlight?: boolean;
   129→  spotlightColor?: string;
   130→}) {
   131→  const inner = (
   132→    <CardContent className="relative z-10 p-4">
   133→      <div className="flex items-start justify-between gap-3">
   134→        <div className="min-w-0">
   135→          <h2 className="text-sm font-medium">{title}</h2>
   136→          {subtitle && <p className="mt-0.5 text-xs text-muted-foreground">{subtitle}</p>}
   137→        </div>
   138→        {actions && <div className="shrink-0">{actions}</div>}
   139→      </div>
   140→      {children}
   141→    </CardContent>
   142→  );
   143→  if (spotlight) {
   144→    return (
   145→      <SpotlightCard className={cn("gap-0 p-0", className)} spotlightColor={spotlightColor}>
   146→        {inner}
   147→      </SpotlightCard>
   148→    );
   149→  }
   150→  return (
   151→    <Card className={cn("lm-panel gap-0 p-0", className)}>
   152→      {inner}
   153→    </Card>
   154→  );
   155→}

   168→export function DashboardView({ session }: { session: SessionData }) {

   175→
   176→  useEffect(() => {
   177→    let cancelled = false;

   181→        const [dashboard, trends] = await Promise.all([

   188→      } catch (err) {
   189→        if (cancelled) return;
   190→        setError(err instanceof Error ? err.message : "Failed to load dashboard data");

   192→        if (!cancelled) setPending(false);
   193→      }
   194→    }
   195→    void run();

   197→      cancelled = true;
   198→    };
   199→  }, [refreshKey]);
   200→
   201→  const refresh = () => setRefreshKey((k) => k + 1);
   202→
   203→  const data = payload?.data ?? null;
   204→  const weeks = payload?.weeks ?? null;
   205→  const refreshing = pending && payload !== null;
   206→
   207→  const pipelineData = useMemo<PipelineDatum[]>(() => {
   208→    if (!data) return [];

   210→      { name: "Sales Ready", value: data.counts.salesReady, color: "#10b981" },
   211→      { name: "Nurture", value: data.counts.nurture, color: "#f59e0b" },
   212→      { name: "Keep Warm", value: data.counts.keepWarm, color: "#a1a1aa" },
   213→      { name: "Manual Review", value: data.counts.manualReview, color: "#f43f5e" },
   214→    ];
   215→  }, [data]);
   216→
   217→  const sortedIndustries = useMemo(() => {
   218→    if (!data) return [];
   219→    return [...data.industries].sort((a, b) => {
   220→      if (a.conversionRate === null && b.conversionRate === null) return b.leads - a.leads;
   221→      if (a.conversionRate === null) return 1;
   222→      if (b.conversionRate === null) return -1;
   223→      return b.conversionRate - a.conversionRate;
   224→    });
   225→  }, [data]);
   226→
   227→  // AI business insights — derived strictly from the live dashboard payload
   228→  const insights = useMemo<Insight[]>(() => {
   229→    if (!data) return [];
   230→    const list: Insight[] = [];
   231→    const total = data.counts.leads;
   232→    if (total > 0) {
   233→      const readyShare = Math.round((data.counts.salesReady / total) * 100);
   234→      list.push({
   235→        id: "ready",
   236→        icon: Target,
   237→        tone: "success",
   238→        text: `${data.counts.salesReady} of ${total} leads (${readyShare}%) are Sales Ready — ${
   239→          readyShare >= 25 ? "a strong active pipeline" : "worth reviewing nurture thresholds"
   240→        }.`,
   241→      });
   242→    }
   243→    if (data.counts.predictions > 0) {
   244→      list.push({
   245→        id: "pred",
   246→        icon: Gauge,
   247→        tone: "brand",
   248→        text: `The model predicts ${data.conversionEstimate.predictedWinners} likely conversions across ${data.counts.predictions} scored leads${
   249→          data.averages.intent !== null ? `, with average buying intent at ${Math.round(data.averages.intent)}/100` : ""
   250→        }.`,
   251→      });
   252→    }
   253→    const bestIndustry = sortedIndustries.find((i) => i.conversionRate !== null && i.won > 0);
   254→    if (bestIndustry) {
   255→      list.push({
   256→        id: "ind",
   257→        icon: Factory,
   258→        tone: "info",
   259→        text: `${bestIndustry.industry} converts best so far (${fmtConversion(bestIndustry.conversionRate)} from ${bestIndustry.won} won) — prioritise similar verticals.`,
   260→      });
   261→    }
   262→    const topRec = data.topRecommendations[0];
   263→    if (topRec) {
   264→      list.push({
   265→        id: "rec",
   266→        icon: Sparkles,
   267→        tone: "warning",
   268→        text: `Highest-priority next step: ${prettifyAction(topRec.action).toLowerCase()} at ${topRec.companyName}.`,
   269→      });
   270→    }
   271→    return list.slice(0, 4);
   272→  }, [data, sortedIndustries]);
   273→
   274→  return (

   281→          <>
   282→            {data && (

   284→                <span
   285→                  className={cn(
   286→                    "h-1.5 w-1.5 rounded-full",
   287→                    data.llmMode === "live" ? "bg-emerald-500 lm-pulse-dot" : "bg-zinc-400",
   288→                  )}
   289→                  aria-hidden="true"
   290→                />
   291→                {data.llmMode === "live" ? "LLM live" : "LLM demo"}
   292→              </Badge>
   293→            )}
   294→            <Button variant="outline" size="sm" onClick={refresh} disabled={pending} aria-label="Refresh dashboard data">
   295→              <RefreshCw className={cn("size-4", refreshing && "animate-spin")} aria-hidden="true" />
   296→              Refresh
   297→            </Button>
   298→          </>
   299→        }
   300→      />
   301→
   302→      {error && data && (
   303→        <Alert variant="destructive">
   304→          <TriangleAlert aria-hidden="true" />

   308→            <Button variant="outline" size="sm" onClick={refresh} aria-label="Retry dashboard refresh">
   309→              Retry

   311→          </AlertDescription>
   312→        </Alert>

   314→
   315→      {pending && !data ? (

   322→            <span>{error}</span>
   323→            <Button variant="outline" size="sm" onClick={refresh} aria-label="Retry loading dashboard">
   324→              Retry
   325→            </Button>
   326→          </AlertDescription>
   327→        </Alert>

   333→            <StatCard title="Sales Ready" value={data.counts.salesReady} tone="brand" icon={<Target className="size-3.5" />} />

   388→            <PanelCard title="Model status" subtitle="Conversion prediction engine" spotlight spotlightColor="var(--lm-info-soft)">
