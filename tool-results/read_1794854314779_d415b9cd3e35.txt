     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import type { SessionData } from "@/lib/store";
     5→import { useAppStore } from "@/lib/store";
     6→import { api, fmtDateTime, fmtRelative } from "@/lib/api";
     7→import { Card, CardContent } from "@/components/ui/card";
     8→import { Button } from "@/components/ui/button";
     9→import { Badge } from "@/components/ui/badge";
    10→import { Textarea } from "@/components/ui/textarea";
    11→import { ScrollArea } from "@/components/ui/scroll-area";
    12→import { Separator } from "@/components/ui/separator";
    13→import { Skeleton } from "@/components/ui/skeleton";
    14→import {
    15→  BrainCircuit,
    16→  Check,
    17→  ChevronDown,
    18→  Copy,
    19→  Loader2,
    20→  MessageSquarePlus,
    21→  RefreshCcw,
    22→  Send,
    23→  Trash2,
    24→  User,
    25→  Wrench,
    26→  X,
    27→  Zap,
    28→} from "lucide-react";
    29→import { toast } from "sonner";
    30→import { cn } from "@/lib/utils";
    31→import { Markdown, EmptyState } from "@/components/lm/shared";
    32→import { LiquidGlassButton } from "@/components/ui/liquid-glass-button";
    33→
    34→interface ToolTrace {
    35→  name: string;
    36→  args: Record<string, unknown>;
    37→  ok: boolean;
    38→  summary: string;
    39→  ms: number;
    40→}
    41→
    42→interface ChatMsg {
    43→  id: string;
    44→  role: "USER" | "ASSISTANT";
    45→  content: string;
    46→  toolTraces: ToolTrace[];
    47→  createdAt: string;
    48→  streaming?: boolean;
    49→}
    50→
    51→interface ConversationSummary {
    52→  id: string;
    53→  title: string;
    54→  mode: string;
    55→  updatedAt: string;
    56→  firstMessage: string;
    57→}
    58→
    59→interface PendingCard {
    60→  pendingActionId: string;
    61→  summary: string;
    62→}
    63→
    64→export function CopilotView({ session }: { session: SessionData }) {
    65→  const conversationId = useAppStore((s) => s.conversationId);
    66→  const leadContextId = useAppStore((s) => s.leadContextId);
    67→  const navigate = useAppStore((s) => s.navigate);
    68→
    69→  const [conversations, setConversations] = useState<ConversationSummary[]>([]);
    70→  const [messages, setMessages] = useState<ChatMsg[]>([]);
    71→  const [loadingThread, setLoadingThread] = useState(false);
    72→  const [sending, setSending] = useState(false);
    73→  const [input, setInput] = useState("");
    74→  const [statusLine, setStatusLine] = useState<string | null>(null);
    75→  const [pending, setPending] = useState<PendingCard | null>(null);
    76→  const [suggestions, setSuggestions] = useState<string[]>([]);
    77→  const [contextLead, setContextLead] = useState<{ id: string; label: string } | null>(null);
    78→  const [mobileListOpen, setMobileListOpen] = useState(false);
    79→  const bottomRef = useRef<HTMLDivElement>(null);
    80→  const abortRef = useRef<AbortController | null>(null);
    81→  // When send() creates a fresh conversation we optimistically append messages;
    82→  // the conversationId effect must not clobber them with an empty server thread.
    83→  const skipThreadLoadRef = useRef(false);
    84→
    85→  const loadConversations = useCallback(async () => {
    86→    try {
    87→      const res = await api<{ conversations: ConversationSummary[] }>("/api/chat/conversations");
    88→      setConversations(res.conversations);
    89→    } catch {
    90→      // sidebar is non-critical
    91→    }
    92→  }, []);
    93→
    94→  useEffect(() => {
    95→    loadConversations();
    96→  }, [loadConversations]);
    97→
    98→  useEffect(() => {
    99→    api<{ suggestions: string[] }>(`/api/chat/suggestions?page=copilot${leadContextId ? `&leadId=${leadContextId}` : ""}`)
   100→      .then((r) => setSuggestions(r.suggestions))
   101→      .catch(() => setSuggestions([]));
   102→  }, [leadContextId]);
   103→
   104→  useEffect(() => {
   105→    if (!leadContextId) {
   106→      setContextLead(null);
   107→      return;
   108→    }
   109→    api<{ detail: { lead: { contactName: string; company: { name: string } | null } } }>(`/api/leads/${leadContextId}`)
   110→      .then((r) => setContextLead({ id: leadContextId, label: `${r.detail.lead.contactName} — ${r.detail.lead.company?.name ?? "unknown"}` }))
   111→      .catch(() => setContextLead(null));
   112→  }, [leadContextId]);
   113→
   114→  const loadThread = useCallback(async (id: string) => {
   115→    setLoadingThread(true);
   116→    setPending(null);
   117→    try {
   118→      const res = await api<{
   119→        conversation: { id: string; title: string; mode: string; messages: { id: string; role: "USER" | "ASSISTANT"; content: string; toolTraces: ToolTrace[]; createdAt: string }[] };
   120→      }>(`/api/chat/conversations/${id}`);
   121→      setMessages(res.conversation.messages);
   122→    } catch (e) {
   123→      toast.error(e instanceof Error ? e.message : "Failed to load conversation");
   124→    } finally {
   125→      setLoadingThread(false);
   126→    }
   127→  }, []);
   128→
   129→  useEffect(() => {
   130→    if (conversationId) {
   131→      if (skipThreadLoadRef.current) {
   132→        skipThreadLoadRef.current = false;
   133→        return;
   134→      }
   135→      loadThread(conversationId);
   136→    } else {
   137→      setMessages([]);
   138→    }
   139→  }, [conversationId, loadThread]);
   140→
   141→  useEffect(() => {
   142→    bottomRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
   143→  }, [messages, statusLine]);
   144→
   145→  const newConversation = async () => {
   146→    try {
   147→      const res = await api<{ conversation: { id: string } }>("/api/chat/conversations", { method: "POST", body: {} });
   148→      await loadConversations();
   149→      navigate("copilot", { conversationId: res.conversation.id });
   150→    } catch (e) {
   151→      toast.error(e instanceof Error ? e.message : "Could not create conversation");
   152→    }
   153→  };
   154→
   155→  const deleteConversation = async (id: string) => {
   156→    try {
   157→      await api(`/api/chat/conversations/${id}`, { method: "DELETE" });
   158→      if (conversationId === id) {
   159→        navigate("copilot");
   160→        setMessages([]);
   161→      }
   162→      await loadConversations();
   163→      toast.success("Conversation deleted");
   164→    } catch (e) {
   165→      toast.error(e instanceof Error ? e.message : "Delete failed");
   166→    }
   167→  };
   168→
   169→  // ── Streaming send ─────────────────────────────────────────────────────────
   170→
   171→  const send = async (content: string) => {
   172→    if (!content.trim() || sending) return;
   173→    let convId = conversationId;
   174→    if (!convId) {
   175→      try {
   176→        const res = await api<{ conversation: { id: string } }>("/api/chat/conversations", { method: "POST", body: {} });
   177→        convId = res.conversation.id;
   178→        skipThreadLoadRef.current = true;
   179→        navigate("copilot", { conversationId: convId });
   180→        await loadConversations();
   181→      } catch (e) {
   182→        toast.error(e instanceof Error ? e.message : "Could not create conversation");
   183→        return;
   184→      }
   185→    }
   186→    setSending(true);
   187→    setPending(null);
   188→    setMessages((m) => [
   189→      ...m,
   190→      { id: `local-${Date.now()}`, role: "USER", content, toolTraces: [], createdAt: new Date().toISOString() },
   191→      { id: "streaming", role: "ASSISTANT", content: "", toolTraces: [], createdAt: new Date().toISOString(), streaming: true },
   192→    ]);
   193→    setInput("");
   194→    setStatusLine("Thinking…");
   195→
   196→    const controller = new AbortController();
   197→    abortRef.current = controller;
   198→
   199→    try {
   200→      const res = await fetch(`/api/chat/conversations/${convId}/messages`, {
   201→        method: "POST",
   202→        headers: { "Content-Type": "application/json" },
   203→        body: JSON.stringify({ content, leadContextId: leadContextId ?? undefined }),
   204→        signal: controller.signal,
   205→      });
   206→      if (!res.ok || !res.body) {
   207→        const err = (await res.json().catch(() => ({}))) as { error?: string };
   208→        throw new Error(err.error ?? `Chat failed (${res.status})`);
   209→      }
   210→      const reader = res.body.getReader();
   211→      const decoder = new TextDecoder();
   212→      let buf = "";
   213→      let answerDone = false;
   214→
   215→      const handleEvent = (ev: Record<string, unknown>) => {
   216→        const type = ev.type as string;
   217→        if (type === "status") {
   218→          const stage = ev.stage as string;
   219→          if (stage === "executing_tool") setStatusLine(`Running tool: ${ev.tool}`);
   220→          else if (stage === "answering") setStatusLine(null);
   221→          else if (stage === "finalizing") setStatusLine("Finalizing answer…");
   222→          else if (stage === "demo_reasoning") setStatusLine("Demo assistant analysing workspace data…");
   223→          else if (stage?.includes("falling_back")) setStatusLine("External LLM unavailable — using local demo assistant…");
   224→          else setStatusLine("Thinking…");
   225→        } else if (type === "tools") {
   226→          const traces = ev.traces as ToolTrace[];
   227→          setStatusLine(null);
   228→          setMessages((m) =>
   229→            m.map((msg) => (msg.id === "streaming" ? { ...msg, toolTraces: [...msg.toolTraces, ...traces] } : msg)),
   230→          );
   231→        } else if (type === "delta") {
   232→          setStatusLine(null);
   233→          const text = ev.text as string;
   234→          setMessages((m) => m.map((msg) => (msg.id === "streaming" ? { ...msg, content: msg.content + text } : msg)));
   235→        } else if (type === "pending_action") {
   236→          setPending({ pendingActionId: ev.pendingActionId as string, summary: ev.summary as string });
   237→        } else if (type === "done") {
   238→          answerDone = true;
   239→          setStatusLine(null);
   240→          const finalId = (ev.messageId as string) ?? `done-${Date.now()}`;
   241→          const traces = (ev.traces as ToolTrace[]) ?? [];
   242→          setMessages((m) =>
   243→            m.map((msg) => (msg.id === "streaming" ? { ...msg, id: finalId, toolTraces: traces, streaming: false } : msg)),
   244→          );
   245→          if (ev.title) loadConversations();
   246→        } else if (type === "error") {
   247→          answerDone = true;
   248→          setStatusLine(null);
   249→          setMessages((m) =>
   250→            m.map((msg) =>
   251→              msg.id === "streaming"
   252→                ? { ...msg, id: `err-${Date.now()}`, streaming: false, content: msg.content || `**Error:** ${ev.message}` }
   253→                : msg,
   254→            ),
   255→          );
   256→          toast.error((ev.message as string) ?? "Chat error");
   257→        }
   258→      };
   259→
   260→      while (true) {
   261→        const { done, value } = await reader.read();
   262→        if (done) break;
   263→        buf += decoder.decode(value, { stream: true });
   264→        let idx: number;
   265→        while ((idx = buf.indexOf("\n\n")) >= 0) {
   266→          const raw = buf.slice(0, idx).trim();
   267→          buf = buf.slice(idx + 2);
   268→          if (raw.startsWith("data:")) {
   269→            try {
   270→              handleEvent(JSON.parse(raw.slice(5).trim()));
   271→            } catch {
   272→              // malformed event
   273→            }
   274→          }
   275→        }
   276→      }
   277→      if (!answerDone) {
   278→        setMessages((m) => m.map((msg) => (msg.id === "streaming" ? { ...msg, streaming: false } : msg)));
   279→      }
   280→    } catch (e) {
   281→      if ((e as Error).name === "AbortError") {
   282→        setMessages((m) => m.map((msg) => (msg.id === "streaming" ? { ...msg, streaming: false, content: msg.content || "_Stopped._" } : msg)));
   283→      } else {
   284→        const msg = e instanceof Error ? e.message : "Chat failed";
   285→        setMessages((m) => m.map((m2) => (m2.id === "streaming" ? { ...m2, streaming: false, content: `**Error:** ${msg}` } : m2)));
   286→        toast.error(msg);
   287→      }
   288→    } finally {
   289→      setSending(false);
   290→      setStatusLine(null);
   291→      abortRef.current = null;
   292→    }
   293→  };
   294→
   295→  const stop = () => abortRef.current?.abort();
   296→
   297→  const resolvePending = async (action: "confirm" | "cancel") => {
   298→    if (!pending) return;
   299→    try {
   300→      const res = await api<{ result?: { ok: boolean; summary: string }; cancelled?: boolean }>(`/api/chat/pending/${pending.pendingActionId}`, {
   301→        method: "POST",
   302→        body: { action },
   303→      });
   304→      if (action === "confirm") {
   305→        if (res.result?.ok) toast.success(res.result.summary);
   306→        else toast.error(res.result?.summary ?? "Action failed");
   307→        setMessages((m) => [
   308→          ...m,
   309→          {
   310→            id: `sys-${Date.now()}`,
   311→            role: "ASSISTANT",
   312→            content: res.result?.ok
   313→              ? `**Action executed.** ${res.result.summary}`
   314→              : `**Action failed.** ${res.result?.summary ?? "Unknown error"}`,
   315→            toolTraces: [],
   316→            createdAt: new Date().toISOString(),
   317→          },
   318→        ]);
   319→      } else {
   320→        toast.info("Action cancelled — nothing was changed");
   321→        setMessages((m) => [
   322→          ...m,
   323→          { id: `sys-${Date.now()}`, role: "ASSISTANT", content: "**Cancelled.** No changes were made.", toolTraces: [], createdAt: new Date().toISOString() },
   324→        ]);
   325→      }
   326→      setPending(null);
   327→    } catch (e) {
   328→      toast.error(e instanceof Error ? e.message : "Failed to resolve action");
   329→    }
   330→  };
   331→
   332→  const regenerate = async () => {
   333→    const lastUser = [...messages].reverse().find((m) => m.role === "USER");
   334→    if (!lastUser) return;
   335→    setMessages((m) => (m[m.length - 1]?.role === "ASSISTANT" ? m.slice(0, -1) : m));
   336→    await send(lastUser.content);
   337→  };
   338→
   339→  const isDemo = session.llmMode === "demo";
   340→
   341→  return (
   342→    <div className="flex h-[calc(100vh-11.5rem)] min-h-[32rem] flex-col gap-4 lg:flex-row">
   343→      {/* Conversation list */}
   344→      <aside className={cn("w-full shrink-0 lg:w-64", mobileListOpen ? "block" : "hidden lg:block")}>
   345→        <Card className="flex h-full flex-col p-0 gap-0">
   346→          <div className="flex items-center justify-between p-3">
   347→            <p className="text-sm font-medium">Conversations</p>
   348→            <Button size="sm" variant="outline" className="h-8" onClick={newConversation}>
   349→              <MessageSquarePlus className="mr-1 h-3.5 w-3.5" /> New
   350→            </Button>
   351→          </div>
   352→          <Separator />
   353→          <ScrollArea className="h-full max-h-[calc(100vh-16rem)] flex-1 lm-scroll">
   354→            <div className="space-y-1 p-2">
   355→              {conversations.length === 0 && <p className="px-2 py-6 text-center text-xs text-muted-foreground">No conversations yet</p>}
   356→              {conversations.map((c) => (
   357→                <div
   358→                  key={c.id}
   359→                  className={cn(
   360→                    "group flex cursor-pointer items-start justify-between gap-2 rounded-md px-2 py-2 text-sm transition-colors",
   361→                    conversationId === c.id ? "bg-[var(--lm-brand-soft)] text-[var(--lm-brand)]" : "hover:bg-muted",
   362→                  )}
   363→                  onClick={() => {
   364→                    navigate("copilot", { conversationId: c.id });
   365→                    setMobileListOpen(false);
   366→                  }}
   367→                  role="button"
   368→                  tabIndex={0}
   369→                  onKeyDown={(e) => e.key === "Enter" && navigate("copilot", { conversationId: c.id })}
   370→                >
   371→                  <div className="min-w-0">
   372→                    <p className="truncate font-medium">{c.title}</p>
   373→                    <p className="truncate text-xs text-muted-foreground">{fmtRelative(c.updatedAt)} · {c.mode === "DEMO" ? "demo" : "llm"}</p>
   374→                  </div>
   375→                  <button
   376→                    className="rounded p-1 text-muted-foreground opacity-0 transition-opacity hover:text-rose-500 group-hover:opacity-100"
   377→                    aria-label={`Delete conversation ${c.title}`}
   378→                    onClick={(e) => {
   379→                      e.stopPropagation();
   380→                      deleteConversation(c.id);
   381→                    }}
   382→                  >
   383→                    <Trash2 className="h-3.5 w-3.5" />
   384→                  </button>
   385→                </div>
   386→              ))}
   387→            </div>
   388→          </ScrollArea>
   389→        </Card>
   390→      </aside>
   391→
   392→      {/* Chat column */}
   393→      <div className="flex min-w-0 flex-1 flex-col">
   394→        <Card className="flex min-h-0 flex-1 flex-col p-0 gap-0">
   395→          {/* Chat header */}
   396→          <div className="flex items-center justify-between gap-2 border-b px-4 py-2.5">
   397→            <div className="flex min-w-0 items-center gap-2">
   398→              <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-md bg-[var(--lm-brand-soft)] ring-1 ring-[var(--lm-brand)]/30">
   399→                <BrainCircuit className="h-4 w-4 text-[var(--lm-brand)]" />
   400→              </div>
   401→              <div className="min-w-0">
   402→                <p className="truncate text-sm font-medium">LeadMind Copilot</p>
   403→                {contextLead ? (
   404→                  <p className="truncate text-xs text-muted-foreground">
   405→                    Context: {contextLead.label}
   406→                    <button className="ml-2 text-[var(--lm-brand)] hover:underline" onClick={() => navigate("lead", { leadId: contextLead.id })}>
   407→                      open profile
   408→                    </button>
   409→                    <button className="ml-2 text-muted-foreground hover:underline" onClick={() => navigate("copilot")}>
   410→                      clear
   411→                    </button>
   412→                  </p>
   413→                ) : (
   414→                  <p className="truncate text-xs text-muted-foreground">{isDemo ? "Local demo assistant — deterministic, live workspace data" : "Connected to external LLM"}</p>
   415→                )}
   416→              </div>
   417→            </div>
   418→            <div className="flex items-center gap-2">
   419→              <Badge variant="outline" className={cn("text-[10px]", isDemo ? "border-amber-500/40 text-amber-600 dark:text-amber-400" : "border-[var(--lm-brand)]/40 text-[var(--lm-brand)]")}>
   420→                {isDemo ? "DEMO MODE" : "LIVE LLM"}
   421→              </Badge>
   422→              {messages.length > 0 && (
   423→                <Button variant="ghost" size="icon" className="h-8 w-8" onClick={regenerate} disabled={sending} aria-label="Regenerate last answer" title="Regenerate last answer">
   424→                  <RefreshCcw className="h-4 w-4" />
   425→                </Button>
   426→              )}
   427→              <Button variant="ghost" size="icon" className="h-8 w-8 lg:hidden" onClick={() => setMobileListOpen((v) => !v)} aria-label="Toggle conversation list">
   428→                <ChevronDown className="h-4 w-4" />
   429→              </Button>
   430→            </div>
   431→          </div>
   432→
   433→          {/* Messages */}
   434→          <ScrollArea className="min-h-0 flex-1 lm-scroll">
   435→            <div className="space-y-4 px-4 py-4">
   436→              {loadingThread ? (
   437→                <div className="space-y-3">
   438→                  <Skeleton className="h-16 w-2/3" />
   439→                  <Skeleton className="ml-auto h-24 w-3/4" />
   440→                  <Skeleton className="h-16 w-1/2" />
   441→                </div>
   442→              ) : messages.length === 0 ? (
   443→                <div className="py-10">
   444→                  <EmptyState
   445→                    icon={<BrainCircuit className="h-10 w-10" />}
   446→                    title="Ask your pipeline anything"
   447→                    description="The Copilot investigates real workspace data through validated tools — lead search, scoring explanations, analytics, recommendations and knowledge retrieval. It never invents records."
   448→                  />
   449→                  {suggestions.length > 0 && (
   450→                    <div className="mx-auto mt-4 flex max-w-2xl flex-wrap justify-center gap-2">
   451→                      {suggestions.map((s, i) => (
   452→                        <button
   453→                          key={i}
   454→                          onClick={() => send(s)}
   455→                          disabled={sending}
   456→                          className="rounded-full border px-3 py-1.5 text-xs text-muted-foreground transition-colors hover:border-[var(--lm-brand)]/40 hover:text-[var(--lm-brand)]"
   457→                        >
   458→                          {s}
   459→                        </button>
   460→                      ))}
   461→                    </div>
   462→                  )}
   463→                </div>
   464→              ) : (
   465→                messages.map((m) => (
   466→                  <div key={m.id} className={cn("flex gap-3", m.role === "USER" ? "justify-end" : "justify-start")}>
   467→                    {m.role === "ASSISTANT" && (
   468→                      <div className="mt-1 flex h-7 w-7 shrink-0 items-center justify-center rounded-md bg-[var(--lm-brand-soft)] ring-1 ring-[var(--lm-brand)]/30">
   469→                        <BrainCircuit className="h-4 w-4 text-[var(--lm-brand)]" />
   470→                      </div>
   471→                    )}
   472→                    <div className={cn("max-w-[85%] space-y-2 sm:max-w-[80%]", m.role === "USER" && "flex flex-col items-end")}>
   473→                      {m.toolTraces.length > 0 && <ToolTracePanel traces={m.toolTraces} />}
   474→                      <div
   475→                        className={cn(
   476→                          "rounded-2xl px-4 py-3",
   477→                          m.role === "USER"
   478→                            ? "bg-[var(--lm-brand)] text-[var(--primary-foreground)]"
   479→                            : "lm-panel border bg-card",
   480→                          m.streaming && m.content.length === 0 && "animate-pulse",
   481→                        )}
   482→                      >
   483→                        {m.role === "USER" ? (
   484→                          <p className="whitespace-pre-wrap text-sm">{m.content}</p>
   485→                        ) : (
   486→                          <div className={cn(m.streaming && "lm-caret")}>
   487→                            {m.content ? <Markdown content={m.content} /> : <p className="text-sm text-muted-foreground">…</p>}
   488→                          </div>
   489→                        )}
   490→                      </div>
   491→                      <div className="flex items-center gap-2 px-1 text-[10px] text-muted-foreground">
   492→                        <span>{fmtDateTime(m.createdAt)}</span>
   493→                        {m.role === "ASSISTANT" && !m.streaming && m.content.length > 0 && (
   494→                          <button
   495→                            className="inline-flex items-center gap-1 hover:text-foreground"
   496→                            aria-label="Copy response"
   497→                            onClick={() => {
   498→                              navigator.clipboard.writeText(m.content).then(

   589→              )}
   590→            </div>
   591→          </div>

   594→    </div>
