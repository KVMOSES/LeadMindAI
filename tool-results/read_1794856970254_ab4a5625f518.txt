    38→    38→ * This component renders the animated chat surface (message stream, typing

    48→    48→ *   are disabled under `prefers-reduced-motion`.
    49→    49→ * - No attachment controls: the current backend has no file-upload endpoint,

   118→   118→export function expandSlashCommand(text: string): string {
   119→   119→  const match = text.match(/^\/([a-zA-Z]+)\s*([\s\S]*)$/);
   120→   120→  if (!match) return text;
   121→   121→  const base = COMMAND_EXPANSIONS[`/${match[1].toLowerCase()}`];
   122→   122→  if (!base) return text;
   123→   123→  const rest = match[2].trim();
   124→   124→  return rest ? `${base}. Details: ${rest}` : base;
   125→   125→}

   316→   316→
   317→   317→export function AnimatedAiChat({
   318→   318→  messages,
   319→   319→  loadingThread,
   320→   320→  sending,
   321→   321→  statusLine,
   322→   322→  pending,
   323→   323→  suggestions,
   324→   324→  isDemo,
   325→   325→  contextLead,
   326→   326→  onSend,
   327→   327→  onStop,
   328→   328→  onRetry,
   329→   329→  onResolvePending,
   330→   330→  onOpenContextLead,
   331→   331→  onClearContext,
   332→   332→  headerActions,
   333→   333→  className,
   334→   334→}: AnimatedAiChatProps) {
   335→   335→  const reduced = useReducedMotion();
   336→   336→
   337→   337→  const [input, setInput] = React.useState("");
   338→   338→  const textareaRef = React.useRef<HTMLTextAreaElement>(null);
   339→   339→  const scrollRef = React.useRef<HTMLDivElement>(null);
   340→   340→  const bottomRef = React.useRef<HTMLDivElement>(null);
   341→   341→  const [nearBottom, setNearBottom] = React.useState(true);
   342→   342→
   343→   343→  // ── Command palette state ────────────────────────────────────────────────
   344→   344→  const paletteOpen = input.startsWith("/") && !sending;
   345→   345→  const paletteQuery = paletteOpen ? input.slice(1).split(/\s/)[0]?.toLowerCase() ?? "" : "";
   346→   346→  const filteredCommands = React.useMemo(() => {
   347→   347→    if (!paletteOpen) return [];
   348→   348→    if (!paletteQuery) return COPILOT_COMMANDS;
   349→   349→    return COPILOT_COMMANDS.filter((c) => c.command.slice(1).startsWith(paletteQuery));
   350→   350→  }, [paletteOpen, paletteQuery]);
   351→   351→  const [activeCmdIndex, setActiveCmdIndex] = React.useState(0);
   352→   352→
   353→   353→  React.useEffect(() => {
   354→   354→    setActiveCmdIndex(0);
   355→   355→  }, [paletteQuery]);
   356→   356→
   357→   357→  // ── Auto-resizing composer ───────────────────────────────────────────────
   358→   358→  React.useLayoutEffect(() => {
   359→   359→    const el = textareaRef.current;
   360→   360→    if (!el) return;
   361→   361→    el.style.height = "auto";
   362→   362→    el.style.height = `${Math.min(el.scrollHeight, 180)}px`;
   363→   363→  }, [input]);
   364→   364→
   365→   365→  // ── Scroll behaviour: follow only when the user is already at the bottom ──
   366→   366→  const handleScroll = React.useCallback(() => {
   367→   367→    const el = scrollRef.current;
   368→   368→    if (!el) return;
   369→   369→    setNearBottom(el.scrollHeight - el.scrollTop - el.clientHeight < 80);
   370→   370→  }, []);
   371→   371→
   372→   372→  React.useEffect(() => {
   373→   373→    if (nearBottom) {
   374→   374→      bottomRef.current?.scrollIntoView({ behavior: reduced ? "auto" : "smooth", block: "end" });
   375→   375→    }
   376→   376→  }, [messages, statusLine, nearBottom, reduced]);
   377→   377→
   378→   378→  const scrollToLatest = () => {
   379→   379→    bottomRef.current?.scrollIntoView({ behavior: reduced ? "auto" : "smooth", block: "end" });
   380→   380→  };
   381→   381→
   382→   382→  const selectCommand = (cmd: CopilotCommand) => {
   383→   383→    if (cmd.send) {
   384→   384→      setInput("");
   385→   385→      onSend(cmd.send);
   386→   386→      textareaRef.current?.focus();
   387→   387→      return;
   388→   388→    }
   389→   389→    setInput(cmd.template);
   390→   390→    requestAnimationFrame(() => {
   391→   391→      textareaRef.current?.focus();
   392→   392→      const el = textareaRef.current;
   393→   393→      if (el) el.selectionStart = el.selectionEnd = el.value.length;
   394→   394→    });
   395→   395→  };
   396→   396→
   397→   397→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   398→   398→    const trimmedInput = input.trim();
   399→   399→    // Exact command match (e.g. "/pipeline") → Enter sends instead of re-selecting.
   400→   400→    const exactCommand =
   401→   401→      paletteOpen &&
   402→   402→      trimmedInput.length > 1 &&
   403→   403→      filteredCommands.some((c) => c.command === trimmedInput);
   404→   404→
   405→   405→    if (paletteOpen && filteredCommands.length > 0) {
   406→   406→      if (e.key === "ArrowDown") {
   407→   407→        e.preventDefault();
   408→   408→        setActiveCmdIndex((i) => (i + 1) % filteredCommands.length);
   409→   409→        return;
   410→   410→      }
   411→   411→      if (e.key === "ArrowUp") {
   412→   412→        e.preventDefault();
   413→   413→        setActiveCmdIndex((i) => (i - 1 + filteredCommands.length) % filteredCommands.length);
   414→   414→        return;
   415→   415→      }
   416→   416→      if ((e.key === "Enter" || e.key === "Tab") && !exactCommand) {
   417→   417→        e.preventDefault();
   418→   418→        selectCommand(filteredCommands[activeCmdIndex]);
   419→   419→        return;
   420→   420→      }
   421→   421→      if (e.key === "Escape") {
   422→   422→        e.preventDefault();
   423→   423→        setInput("");
   424→   424→        return;
   425→   425→      }
   426→   426→    }
   427→   427→    if (e.key === "Enter" && !e.shiftKey) {
   428→   428→      e.preventDefault();
   429→   429→      if (trimmedInput) {
   430→   430→        setInput("");
   431→   431→        onSend(expandSlashCommand(trimmedInput));
   432→   432→      }
   433→   433→    }
   434→   434→  };
   435→   435→
   436→   436→  const lastMsg = messages[messages.length - 1];
   437→   437→  const lastAssistantErrored = lastMsg?.role === "ASSISTANT" && !lastMsg.streaming && lastMsg.content.startsWith("**Error:**");
   438→   438→  const canRetry = !sending && messages.some((m) => m.role === "USER");
   439→   439→
   440→   440→  return (
